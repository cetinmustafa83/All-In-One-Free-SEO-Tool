     1→//! CSV exports — Task 14.8a (strangler-fig phase 14.8, wave 1).
     2→//!
     3→//! Ports the four golden-path CSV downloads from src/app/*/export.csv/route.ts
     4→//! to `POST /api/internal/exports/*`, byte-for-byte against the original
     5→//! implementation (golden fixtures in backend/fixtures/14.8/ — captured from
     6→//! the Next routes BEFORE the swap and compared after):
     7→//!
     8→//!   clients      → clients.csv       (techStack JSON array → join("|"))
     9→//!   competitors  → competitors.csv   (LEFT JOIN clients, no ?? "" wrappers)
    10→//!   audits       → audits.csv        (nullable score/started/completed → "")
    11→//!   backlinks    → backlinks.csv     (nullable client join cols → "")
    12→//!
    13→//! The Next routes stay at the same URLs and become thin proxies
    14→//! (session → userId, ?clientId/?client → body) — the CSV bytes come from
    15→//! here. Response envelope: {"ok":true,"filename","contentType",
    16→//! "disposition","csv"}; the proxy re-wraps only the transport headers.
    17→//!
    18→//! Serializer parity (src/lib/csv-export.ts):
    19→//!   * null/undefined → ""; numbers → String(n); strings pass through;
    20→//!   * Date → Date.toISOString(): "YYYY-MM-DDTHH:MM:SS.mmmZ" (always 3
    21→//!     millisecond digits — SQLite stores epoch SECONDS, drizzle
    22→//!     mode:"timestamp" renders `new Date(v * 1000)`);
    23→//!   * quote-wrap iff the value matches /[",\n\r]/ (byte scan — the four
    24→//!     triggers are ASCII, so UTF-8 continuation bytes can never hide one),
    25→//!     doubling `"` → `""` (RFC 4180);
    26→//!   * cells joined "," / lines "\n", NO trailing newline, NO BOM, header
    27→//!     row always present even for an empty result.
    28→//!
    29→//! Date formatting is hand-rolled (Howard Hinnant civil_from_days, ~25
    30→//! lines) instead of chrono: byte-exact control of the JS surface — the
    31→//! 4-digit year padding, the ±6-digit signed years outside 0..9999 and the
    32→//! engine's per-component truncation of REAL (fractional-second) storage —
    33→//! none of which chrono's RFC 3339 formatter guarantees.
    34→//!
    35→//! Scope parity (src/lib/client-scope.ts exportScope): see client_scope.rs.
    36→//! `?clientId=` / legacy `?client=` normalization happens on the Next side
    37→//! in TypeScript, where Number()/Number.isInteger() ARE the semantics being
    38→//! mirrored — zero reimplementation risk. The body carries the resolved
    39→//! meaning: absent/null = no filter, positive id = `= id`, anything else
    40→//! (0 / negative / unparseable → the proxy's -1 sentinel) = the always-false
    41→//! `IN (-1)` — the TS route turns every non-positive-integer Number(raw)
    42→//! into exactly that filter ("an unparseable clientId yields an empty
    43→//! export rather than everything").

    58→/// csvEscape: quote-wrap iff the value contains " , \n or \r; double the
    59→/// inner quotes. The /[",\n\r]/ test is a byte scan in Rust — all four
    60→/// trigger bytes are ASCII, and UTF-8 continuation bytes are ≥ 0x80, so a
    61→/// multi-byte character can never contain one.
    62→pub fn csv_escape(s: &str) -> String {
    63→    if s.bytes()
    64→        .any(|b| matches!(b, b'"' | b',' | b'\n' | b'\r'))
    65→    {
    66→        let mut out = String::with_capacity(s.len() + 2);
    67→        out.push('"');
    68→        for c in s.chars() {
    69→            if c == '"' {
    70→                out.push_str("\"\"");
    71→            } else {
    72→                out.push(c);
    73→            }
    74→        }
    75→        out.push('"');
    76→        out
    77→    } else {
    78→        s.to_string()
    79→    }
    80→}

    82→/// JS `String(n)` for a float cell. Rust `{}` on f64 is shortest-round-trip
    83→/// like JS for the plain-decimal range; JS switches to exponent notation at
    84→/// ≥ 1e21 / < 1e-6 where Rust keeps printing digits (documented deviation —
    85→/// no export column can hold such values today).
    86→fn js_float(f: f64) -> String {
    87→    format!("{f}")
    88→}

    90→/// JS `Array.prototype.join` element coercion (null/undefined → "", nested
    91→/// arrays join with ",", objects → "[object Object]").
    92→fn join_element(v: &Value) -> String {
    93→    match v {
    94→        Value::Null => String::new(),
    95→        Value::String(s) => s.clone(),
    96→        Value::Number(n) => n.to_string(),
    97→        Value::Bool(b) => b.to_string(),
    98→        Value::Array(items) => items
    99→            .iter()
   100→            .map(join_element)
   101→            .collect::<Vec<_>>()
   102→            .join(","),
   103→        Value::Object(_) => "[object Object]".into(),
   104→    }
   105→}

   107→/// clients.techStack cell: drizzle `mode:"json"` JSON.parses the TEXT and
   108→/// the route does `Array.isArray(v) ? v.join("|") : ""`. A non-array JSON
   109→/// value (string/number/object) or unparsable text → "" (the route only
   110→/// every sees valid JSON because drizzle would have thrown first).
   111→fn tech_stack_cell(raw: Option<&str>) -> String {
   112→    match raw.and_then(|t| serde_json::from_str::<Value>(t).ok()) {
   113→        Some(Value::Array(items)) => items
   114→            .iter()
   115→            .map(join_element)
   116→            .collect::<Vec<_>>()
   117→            .join("|"),
   118→        _ => String::new(),
   119→    }
   120→}

   126→/// Howard Hinnant civil_from_days (days since 1970-01-01 → y/m/d) — the
   127→/// inverse of days_from_civil, valid across the whole i64 range.
   128→fn civil_from_days(z: i64) -> (i64, u32, u32) {
   129→    let z = z + 719_468;
   130→    let era = if z >= 0 { z } else { z - 146_096 } / 146_097;
   131→    let doe = z - era * 146_097; // [0, 146096]
   132→    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365; // [0, 399]
   133→    let y = yoe + era * 400;
   134→    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100); // [0, 365]
   135→    let mp = (5 * doy + 2) / 153; // [0, 11]
   136→    let d = (doy - (153 * mp + 2) / 5 + 1) as u32; // [1, 31]
   137→    let m = if mp < 10 { mp + 3 } else { mp - 9 } as u32; // [1, 12]
   138→    (y + i64::from(m <= 2), m, d)
   139→}

   141→/// JS Date.toISOString() from an epoch-ms time value. `None` ⇔ the JS Date
   142→/// would be invalid (|t| > 8.64e15 ms): toISOString throws "Invalid time
   143→/// value" there and the original route surfaced that as a 500.
   144→pub fn iso_from_ms(t: i64) -> Option<String> {
   145→    if t.unsigned_abs() > 8_640_000_000_000_000 {
   146→        return None;
   147→    }
   148→    let days = t.div_euclid(86_400_000);
   149→    let sod = t.rem_euclid(86_400_000); // 0..86_399_999 (ECMA modulo: sign of divisor)
   150→    let ms = (sod % 1000) as u32;
   151→    let sec = sod / 1000;
   152→    let (y, mo, d) = civil_from_days(days);
   153→    let (hh, mm, ss) = (sec / 3600, (sec % 3600) / 60, sec % 60);
   154→    let date = if (0..=9999).contains(&y) {
   155→        format!("{y:04}-{mo:02}-{d:02}")
   156→    } else {
   157→        // JS pads the year to 6 digits with an explicit sign outside 0..9999.
   158→        let sign = if y < 0 { '-' } else { '+' };
   159→        format!("{sign}{:06}", y.abs())
   160→    };
   161→    Some(format!("{date}T{hh:02}:{mm:02}:{ss:02}.{ms:03}Z"))
   162→}

   164→/// Drizzle renders a REAL (fractional seconds) as `new Date(v * 1000)` and
   165→/// the engine extracts components per ECMA: SecFromTime = ⌊t/1000⌋,
   166→/// msFromTime = t mod 1000 (floored modulo) truncated to whole ms by the
   167→/// engine's int cast. Rebuilding the time value from those parts matches
   168→/// V8 for every integral-ms Date, including negatives.
   169→fn real_to_ms(f: f64) -> i64 {
   170→    let t = f * 1000.0;
   171→    if !t.is_finite() {
   172→        return i64::MAX; // Date(NaN) → invalid → throws, mirrored as 500
   173→    }
   174→    let secs = (t / 1000.0).floor();
   175→    let ms = t - secs * 1000.0; // 0 ≤ ms < 1000
   176→    (secs * 1000.0 + ms.trunc()) as i64
   177→}

   179→/// A timestamp column cell → CSV text. Err ⇔ the TS route would have thrown
   180→/// (TEXT in a timestamp column: `value * 1000` → NaN → invalid Date).
   181→fn iso_cell(v: ValueRef<'_>) -> Result<String, ()> {
   182→    Ok(match v {
   183→        ValueRef::Null => String::new(), // TS `r.startedAt ?? ""` family
   184→        ValueRef::Integer(i) => i
   185→            .checked_mul(1000)
   186→            .and_then(iso_from_ms)
   187→            .ok_or(())?,
   188→        ValueRef::Real(f) => iso_from_ms(real_to_ms(f)).ok_or(())?,
   189→        ValueRef::Text(_) => return Err(()),
   190→        ValueRef::Blob(_) => return Err(()),
   191→    })
   192→}

   214→/// Serialize rows: escape each cell, join with ",", join lines with "\n".
   215→/// No trailing newline, no BOM — mirrors `lines.join("\n")`.
   216→fn render(header: &[&str], rows: Vec<Vec<String>>) -> String {
   217→    let mut lines = Vec::with_capacity(rows.len() + 1);
   218→    lines.push(
   219→        header
   220→            .iter()
   221→            .map(|h| csv_escape(h))
   222→            .collect::<Vec<_>>()
   223→            .join(","),
   224→    );
   225→    for row in rows {
   226→        lines.push(
   227→            row.iter()
   228→                .map(|c| csv_escape(c))
   229→                .collect::<Vec<_>>()
   230→                .join(","),
   231→        );
   232→    }
   233→    lines.join("\n")
   234→}

   270→pub fn csv_escape_bespoke(s: &str) -> String {
   271→    if s.bytes().any(|b| matches!(b, b'"' | b',' | b'\n')) {
   272→        let mut out = String::with_capacity(s.len() + 2);
   273→        out.push('"');
   274→        for c in s.chars() {
   275→            if c == '"' {
   276→                out.push_str("\"\"");
   277→            } else {
   278→                out.push(c);
   279→            }
   280→        }
   281→        out.push('"');
   282→        out
   283→    } else {
   284→        s.to_string()
   285→    }
   286→}

   288→/// Serialize keywords rows: the route builds `const csvLines =
   289→/// [header.join(",")]` — the header is joined RAW, no csvEscape pass
   290→/// (unlike csvResponse which escapes headers; same bytes for these literal
   291→/// headers, but the mechanism is mirrored exactly).
   292→fn render_bespoke(header: &[&str], rows: Vec<Vec<String>>) -> String {
   293→    let mut lines = Vec::with_capacity(rows.len() + 1);
   294→    lines.push(header.join(","));
   295→    for row in rows {
   296→        lines.push(
   297→            row.iter()
   298→                .map(|c| csv_escape_bespoke(c))
   299→                .collect::<Vec<_>>()
   300→                .join(","),
   301→        );
   302→    }
   303→    lines.join("\n")
   304→}

   306→/// JS Date semantics on an owned ranking cell: `None` marks a value whose
   307→/// `Date.toISOString()` would throw ("Invalid time value") — deferred to
   308→/// render time because the TS fold stores Date objects and only VISIBLE
   309→/// keywords ever render their winning row's checkedAt.
   310→enum Epoch {
   311→    Iso(String),
   312→    Invalid,
   313→}
   314→
   315→fn epoch_cell(v: ValueRef<'_>) -> Epoch {
   316→    match v {
   317→        ValueRef::Null => Epoch::Iso(String::new()), // notNull column; drizzle null → ""
   318→        ValueRef::Integer(i) => i
   319→            .checked_mul(1000)
   320→            .and_then(iso_from_ms)
   321→            .map(Epoch::Iso)
   322→            .unwrap_or(Epoch::Invalid),
   323→        ValueRef::Real(f) => iso_from_ms(real_to_ms(f)).map_or(Epoch::Invalid, Epoch::Iso),
   324→        // TEXT in a timestamp column: `value * 1000` → NaN → Invalid Date →
   325→        // toISOString() throws RangeError at render.
   326→        ValueRef::Text(_) | ValueRef::Blob(_) => Epoch::Invalid,
   327→    }
   328→}

   330→/// Current UTC civil date, `new Date().toISOString().split("T")[0]` —
   331→/// reuses the 14.8a epoch→ISO helper and takes the date-only part.
   332→fn utc_today() -> String {
   333→    let ms = std::time::SystemTime::now()
   334→        .duration_since(std::time::UNIX_EPOCH)
   335→        .map(|d| d.as_millis() as i64)
   336→        .unwrap_or(0);
   337→    match iso_from_ms(ms) {
   338→        Some(iso) => iso.split('T').next().unwrap_or("").to_string(),
   339→        None => String::new(),
   340→    }
   341→}

   343→/// Pragmatic mirror of JS `a.localeCompare(b)` (default locale) for the
   344→/// backlinks disavow domain sort — validated pairwise against Node's ICU on
   345→/// a 370-string [A-Za-z0-9.-] corpus (139k ordered pairs, 0 mismatches):
   346→///   1. primary: case-FOLD (ICU compares letters case-insensitively —
   347→///      "a".localeCompare("B") === -1);
   348→///   2. tertiary, per char: LOWERCASE before UPPERCASE (ICU root collation
   349→///      orders "b" < "B"; the corpus test proves it);
   350→///   3. original bytes (stable-sort tiebreak; JS sort is stable since
   351→///      ES2019, Rust sort_by is stable — insertion order kept on 0).
   352→/// Documented deviation: full ICU collation for NON-ASCII (é/ß/İ….) and for
   353→/// exotic punctuation (ICU ranks '_' before '-' before '.', unlike ASCII
   354→/// byte order) is NOT mirrored — realistic hostnames are ASCII with only
   355→/// letters/digits/dots/hyphens (IDN lives in URLs as punycode), where the
   356→/// mirror is exact. No ICU dependency is acceptable here; recorded honestly.
   357→fn locale_cmp(a: &str, b: &str) -> std::cmp::Ordering {
   358→    let (fa, fb) = (a.to_lowercase(), b.to_lowercase());
   359→    if fa != fb {
   360→        return fa.cmp(&fb);
   361→    }
   362→    for (ca, cb) in a.chars().zip(b.chars()) {
   363→        let (ra, rb) = (u8::from(ca.is_uppercase()), u8::from(cb.is_uppercase()));
   364→        if ra != rb {
   365→            return ra.cmp(&rb);
   366→        }
   367→    }
   368→    a.cmp(&b)
   369→}

   371→/// JS Number(value) for the reports pinned cell: drizzle's boolean column
   372→/// maps driver values with `Number(value) === 1`, then the route does
   373→/// `r.pinned ? "true" : "false"`.
   374→fn js_number(v: ValueRef<'_>) -> f64 {
   375→    match v {
   376→        ValueRef::Null => 0.0, // Number(null) === 0
   377→        ValueRef::Integer(i) => i as f64,
   378→        ValueRef::Real(f) => f,
   379→        ValueRef::Text(t) => std::str::from_utf8(t)
   380→            .ok()
   381→            .and_then(|s| {
   382→                let s = s.trim(); // Number() trims whitespace
   383→                if s.is_empty() {
   384→                    Some(0.0)
   385→                } else {
   386→                    s.parse::<f64>().ok()
   387→                }
   388→            })
   389→            .unwrap_or(f64::NAN),
   390→        ValueRef::Blob(_) => 0.0, // Number(Uint8Array) === 0
   391→    }
   392→}

   398→struct Scope {

   404→/// Port of exportScope(column, req): the requested client filter
   405→/// (`= id` for a positive integer, `IN (-1)` otherwise) AND the visibility
   406→/// filter (`IN (ids)` / empty-assignment `IN (-1)` / nothing at all).
   407→/// `column` is table-qualified exactly like drizzle's generated SQL.
   408→/// Returns the bare conditions so callers that AND extra terms (disavow's
   409→/// `eq(backlinks.status, "disavow")`) can compose them in the same order
   410→/// drizzle's `and()` produces.
   411→fn build_scope_conds(
   412→    db: &crate::db::Db,
   413→    column: &str,
   414→    user_id: Option<i64>,
   415→    client_id: Option<i64>,
   416→) -> (Vec<String>, Vec<Sqv>) {
   417→    let mut conds: Vec<String> = Vec::new();
   418→    let mut params: Vec<Sqv> = Vec::new();
   419→    if let Some(id) = client_id {
   420→        // TS: Number.isInteger(id) && id > 0 ? eq(column, id) : inArray(column, [-1])
   421→        // The proxy only sends integers; a non-positive one (0 / negative /
   422→        // its -1 sentinel for unparseable raw params) is the always-false arm.
   423→        if id > 0 {
   424→            conds.push(format!("{column} = ?"));
   425→            params.push(Sqv::Integer(id));
   426→        } else {
   427→            conds.push(format!("{column} IN (-1)"));
   428→        }
   429→    }
   430→    if let Some(ids) = visible_client_ids(db, user_id) {
   431→        if ids.is_empty() {
   432→            conds.push(format!("{column} IN (-1)"));
   433→        } else {
   434→            let list = ids.iter().map(|_| "?").collect::<Vec<_>>().join(", ");
   435→            conds.push(format!("{column} IN ({list})"));
   436→            params.extend(ids.into_iter().map(Sqv::Integer));
   437→        }
   438→    }
   439→    (conds, params)
   440→}
   441→
   442→fn build_scope(
   443→    db: &crate::db::Db,
   444→    column: &str,
   445→    user_id: Option<i64>,
   446→    client_id: Option<i64>,
   447→) -> Scope {
   448→    let (conds, params) = build_scope_conds(db, column, user_id, client_id);
   449→    let where_sql = if conds.is_empty() {
   450→        String::new()
   451→    } else {
   452→        format!(" WHERE {}", conds.join(" AND "))
   453→    };
   454→    Scope { where_sql, params }
   455→}

   468→/// Run one export query on the shared connection, mapping each row to its
   469→/// CSV cells through `map_row`. Errors collapse to strings for the 500
   470→/// envelope (bare sqlite message, or "Invalid time value" where the TS
   471→/// route's Date.toISOString() would have thrown).
   472→fn run_export(
   473→    db: &crate::db::Db,
   474→    sql: &str,
   475→    params: &[Sqv],
   476→    map_row: impl Fn(&rusqlite::Row<'_>) -> Result<Vec<String>, String>,
   477→) -> Result<Vec<Vec<String>>, String> {
   478→    let inner = db.with_conn(|c| {
   479→        Ok((|| -> Result<Vec<Vec<String>>, String> {
   480→            let mut stmt = c.prepare(sql).map_err(|e| e.to_string())?;
   481→            let mut q = stmt
   482→                .query(rusqlite::params_from_iter(params.iter()))
   483→                .map_err(|e| e.to_string())?;
   484→            let mut out: Vec<Vec<String>> = Vec::new();
   485→            while let Some(r) = q.next().map_err(|e| e.to_string())? {
   486→                out.push(map_row(r)?);
   487→            }
   488→            Ok(out)
   489→        })())
   490→    });
   491→    inner.map_err(|e| e.to_string()).flatten()
   492→}

  1013→    let scope = build_scope(&state.db, "keywords.client_id", user_id, client_id);
  1014→    let sql = format!(
  1015→        "SELECT keywords.id, keywords.query, keywords.country, keywords.device, \
  1016→         clients.name, clients.url, keywords.created_at \
  1017→         FROM keywords LEFT JOIN clients ON keywords.client_id = clients.id{} \
  1018→         ORDER BY keywords.created_at DESC",
  1019→        scope.where_sql
  1020→    );
  1021→    let params = scope.params.clone();
  1022→    let kw_rows = run_export(&state.db, &sql, &params, |r| {
  1023→        Ok(vec![
  1024→            cell(r.get_ref(0).map_err(|e| e.to_string())?),
  1025→            cell(r.get_ref(1).map_err(|e| e.to_string())?),
  1026→            cell(r.get_ref(2).map_err(|e| e.to_string())?),
  1027→            cell(r.get_ref(3).map_err(|e| e.to_string())?),
  1028→            cell(r.get_ref(4).map_err(|e| e.to_string())?), // clientName ?? ""
  1029→            cell(r.get_ref(5).map_err(|e| e.to_string())?), // clientUrl ?? ""
  1030→            match r.get_ref(6).map_err(|e| e.to_string())? {
  1031→                // createdAt is notNull; render at fold exit like iso_cell would.
  1032→                ValueRef::Integer(i) => i
  1033→                    .checked_mul(1000)
  1034→                    .and_then(iso_from_ms)
  1035→                    .ok_or_else(|| "Invalid time value".to_string())?,
  1036→                ValueRef::Real(f) => iso_from_ms(real_to_ms(f))
  1037→                    .ok_or_else(|| "Invalid time value".to_string())?,
  1038→                ValueRef::Null => String::new(),
  1039→                // TEXT in a timestamp column: `value * 1000` → NaN → the
  1040→                // route's r.createdAt.toISOString() throws RangeError → 500.
  1041→                _ => return Err("Invalid time value".to_string()),
  1042→            },
  1043→        ])
  1044→    });
  1045→    let kw_rows = match kw_rows {
  1046→        Ok(rows) => rows,
  1047→        Err(msg) => return db_fail(&msg),
  1048→    };
  1049→
  1050→    // Full keyword_rankings scan (NO scope filter — the TS route queries the
  1051→    // whole table) ordered by checked_at ASC; fold into a map keeping the
  1052→    // LAST row per keyword — exactly `latest.set(r.keywordId, {...})` in an
  1053→    // ascending scan (Map.set overwrite; the winner is the last scan row,
  1054→    // so equal checked_at resolves to scan order, identically on both
  1055→    // stacks because the engine and statement shape are the same).
  1056→    let fold: Result<std::collections::HashMap<i64, Latest>, String> = state
  1057→        .db
  1058→        .with_conn(|c| {
  1059→            Ok((|| -> Result<std::collections::HashMap<i64, Latest>, String> {
  1060→            let mut stmt = c
  1061→                .prepare(
  1062→                    "SELECT keyword_id, position, url, checked_at FROM keyword_rankings                      ORDER BY keyword_rankings.checked_at",
  1063→                )
  1064→                .map_err(|e| e.to_string())?;
  1065→            let mut q = stmt.query([]).map_err(|e| e.to_string())?;
  1066→            let mut map: std::collections::HashMap<i64, Latest> = std::collections::HashMap::new();
  1067→            while let Some(r) = q.next().map_err(|e| e.to_string())? {
  1068→                let kid: i64 = r.get(0).map_err(|e| e.to_string())?;
  1069→                map.insert(
  1070→                    kid,
  1071→                    Latest {
  1072→                        position: cell(r.get_ref(1).map_err(|e| e.to_string())?),
  1073→                        url: cell(r.get_ref(2).map_err(|e| e.to_string())?),
  1074→                        checked: epoch_cell(r.get_ref(3).map_err(|e| e.to_string())?),
  1075→                    },
  1076→                );
  1077→            }
  1078→            Ok(map)
  1079→            })())
  1080→        })
  1081→        .map_err(|e| e.to_string())
  1082→        .flatten();
  1083→    let latest = match fold {
  1084→        Ok(map) => map,
  1085→        Err(msg) => return db_fail(&msg),
  1086→    };
  1087→
  1088→    let mut rows: Vec<Vec<String>> = Vec::with_capacity(kw_rows.len());
  1089→    for k in &kw_rows {
  1090→        let kid: i64 = match k[0].parse() {
  1091→            Ok(v) => v,
  1092→            Err(_) => return db_fail("bad keyword id"),
  1093→        };
  1094→        let lp = latest.get(&kid);
  1095→        let invalid = || "Invalid time value".to_string();
  1096→        // lp?.checkedAt?.toISOString() ?? "" — the fold value exists but the
  1097→        // winning row's checkedAt is what renders (its own toISOString).
  1098→        let checked = match lp.map(|l| &l.checked) {
  1099→            None => String::new(), // no ranking rows at all → undefined → ""
  1100→            Some(Epoch::Iso(s)) => s.clone(),
  1101→            Some(Epoch::Invalid) => return db_fail(&invalid()),
  1102→        };
  1103→        rows.push(vec![
  1104→            k[4].clone(), // clientName ?? ""
  1105→            k[5].clone(), // clientUrl ?? ""
  1106→            k[1].clone(), // query
  1107→            k[2].clone(), // country
  1108→            k[3].clone(), // device
  1109→            lp.map(|l| l.position.clone()).unwrap_or_default(), // ?? ""
  1110→            lp.map(|l| l.url.clone()).unwrap_or_default(),      // ?? ""
  1111→            checked,
  1112→            k[6].clone(), // createdAt.toISOString()
  1113→        ]);
  1114→    }
  1115→
  1116→    let filename = format!("keywords-{}.csv", utc_today());
  1117→    csv_envelope_dyn(filename, render_bespoke(HEADER, rows))
  1118→}

  1137→pub async fn disavow(State(state): State<AppState>, bytes: Bytes) -> Response {
  1138→    let body: Value = serde_json::from_slice(&bytes).unwrap_or(Value::Null);
  1139→    let (user_id, client_id) = body_ids(&body);
  1140→    // The FILENAME is raw-query-string semantics (`?clientId=` truthiness +
  1141→    // `replace(/[^0-9]/g, "")` — "?clientId=abc" yields disavow-client-.txt),
  1142→    // so the proxy computes it in TypeScript where that regex IS the
  1143→    // semantics, and the body carries it. Default = the no-param branch.
  1144→    let filename = body
  1145→        .get("filename")
  1146→        .and_then(Value::as_str)
  1147→        .unwrap_or("disavow-all.txt")
  1148→        .to_string();
  1149→
  1150→    // drizzle `and(eq(backlinks.status, "disavow"), scope)` — the scope term
  1151→    // is only present when the filter/visibility produced a condition.
  1152→    let (mut conds, mut params) = build_scope_conds(&state.db, "backlinks.client_id", user_id, client_id);
  1153→    conds.insert(0, "backlinks.status = 'disavow'".to_string());
  1154→    let sql = format!(
  1155→        "SELECT backlinks.source_domain, backlinks.notes \
  1156→         FROM backlinks LEFT JOIN clients ON backlinks.client_id = clients.id WHERE {}",
  1157→        conds.join(" AND ")
  1158→    );
  1159→    // No ORDER BY — the TS route scans without one; both stacks therefore
  1160→    // observe the engine's natural rowid scan order, which is what the
  1161→    // grouping (first-link notes) and the stable sort tie on.
  1162→    let scanned: Result<Vec<(Option<String>, Option<String>)>, String> = state
  1163→        .db
  1164→        .with_conn(|c| {
  1165→            Ok((|| -> Result<Vec<(Option<String>, Option<String>)>, String> {
  1166→            let mut stmt = c.prepare(&sql).map_err(|e| e.to_string())?;
  1167→            let mut q = stmt
  1168→                .query(rusqlite::params_from_iter(params.drain(..)))
  1169→                .map_err(|e| e.to_string())?;
  1170→            let mut out = Vec::new();
  1171→            while let Some(r) = q.next().map_err(|e| e.to_string())? {
  1172→                let domain: Option<String> = r.get(0).map_err(|e| e.to_string())?;
  1173→                let notes: Option<String> = r.get(1).map_err(|e| e.to_string())?;
  1174→                out.push((domain, notes));
  1175→            }
  1176→            Ok(out)
  1177→            })())
  1178→        })
  1179→        .map_err(|e| e.to_string())
  1180→        .flatten();
  1181→    let rows = match scanned {
  1182→        Ok(rows) => rows,
  1183→        Err(msg) => return db_fail(&msg),
  1184→    };
  1185→
  1186→    // Group by domain preserving first-encounter order. The Map key is the
  1187→    // string sourceDomain — a NULL sourceDomain makes `a.localeCompare(b)`
  1188→    // throw TypeError on the sort comparator in TS (null has no method),
  1189→    // surfaced here as the same 500.
  1190→    let mut domains: Vec<(Option<String>, String)> = Vec::new(); // (key, first link's notes)
  1191→    for (domain, notes) in &rows {
  1192→        if domains.iter().any(|(k, _)| k == domain) {
  1193→            continue; // later links only append to links[] — links[0] never changes
  1194→        }
  1195→        domains.push((domain.clone(), notes.clone().unwrap_or_default()));
  1196→    }
  1197→    if domains.iter().any(|(k, _)| k.is_none()) {
  1198→        return db_fail("Cannot read properties of null (reading 'localeCompare')");
  1199→    }
  1200→
  1201→    // Sort domains alphabetically — JS localeCompare semantics (locale_cmp).
  1202→    let mut sorted = domains;
  1203→    sorted.sort_by(|(a, _), (b, _)| {
  1204→        let (sa, sb) = (
  1205→            a.as_deref().unwrap_or(""),
  1206→            b.as_deref().unwrap_or(""),
  1207→        );
  1208→        locale_cmp(sa, sb)
  1209→    });
  1210→
  1211→    let mut lines: Vec<String> = vec![
  1212→        "# Google Disavow file generated by SEO tool".to_string(),
  1213→        format!("# Generated: {}", iso_from_ms(now_ms()).unwrap_or_default()),
  1214→        format!("# {} entries across {} domains", rows.len(), sorted.len()),
  1215→        "#".to_string(),
  1216→        "# Submit at: https://search.google.com/search-console/disavow-links-tool".to_string(),
  1217→        "# Only use this if you've identified clearly spammy/manipulative links.".to_string(),
  1218→        "# Most modern sites should NOT submit a disavow file unless under manual action.".to_string(),
  1219→        String::new(),
  1220→    ];
  1221→    for (domain, first_notes) in &sorted {
  1222→        // links[0]?.notes truthiness: null and "" both skip the note line.
  1223→        // `notes.replace(/[\r\n]+/g, " ")` collapses each RUN to one space.
  1224→        let notes = first_notes.as_str();
  1225→        if !notes.is_empty() {
  1226→            lines.push(format!("# {}", collapse_newlines(notes)));
  1227→        }
  1228→        lines.push(format!("domain:{}", domain.as_deref().unwrap_or("null")));
  1229→        lines.push(String::new());
  1230→    }
  1231→
  1232→    envelope(&filename, "text/plain; charset=utf-8", lines.join("\n"))
  1233→}

  1235→/// `now_ms` for the `# Generated:` line — JS new Date().toISOString() of the
  1236→/// wall clock (fixture compares normalize this line; the format must match:
  1237→/// 24-char UTC ISO-8601).
  1238→fn now_ms() -> i64 {
  1239→    std::time::SystemTime::now()
  1240→        .duration_since(std::time::UNIX_EPOCH)
  1241→        .map(|d| d.as_millis() as i64)
  1242→        .unwrap_or(0)
  1243→}
  1244→
  1245→/// JS `s.replace(/[\r\n]+/g, " ")` — each maximal run of CR/LF → one space.
  1246→fn collapse_newlines(s: &str) -> String {
  1247→    let mut out = String::with_capacity(s.len());
  1248→    let mut in_run = false;
  1249→    for c in s.chars() {
  1250→        if c == '\r' || c == '\n' {
  1251→            if !in_run {
  1252→                out.push(' ');
  1253→                in_run = true;
  1254→            }
  1255→        } else {
  1256→            in_run = false;
  1257→            out.push(c);
  1258→        }
  1259→    }
  1260→    out
  1261→}
