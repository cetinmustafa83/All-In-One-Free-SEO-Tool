    58→    58→/// csvEscape: quote-wrap iff the value contains " , \n or \r; double the
    59→    59→/// inner quotes. The /[",\n\r]/ test is a byte scan in Rust — all four
    60→    60→/// trigger bytes are ASCII, and UTF-8 continuation bytes are ≥ 0x80, so a
    61→    61→/// multi-byte character can never contain one.
    62→    62→pub fn csv_escape(s: &str) -> String {
    63→    63→    if s.bytes()
    64→    64→        .any(|b| matches!(b, b'"' | b',' | b'\n' | b'\r'))
    65→    65→    {
    66→    66→        let mut out = String::with_capacity(s.len() + 2);
    67→    67→        out.push('"');
    68→    68→        for c in s.chars() {
    69→    69→            if c == '"' {
    70→    70→                out.push_str("\"\"");
    71→    71→            } else {
    72→    72→                out.push(c);
    73→    73→            }
    74→    74→        }
    75→    75→        out.push('"');
    76→    76→        out
    77→    77→    } else {
    78→    78→        s.to_string()
    79→    79→    }
    80→    80→}

    90→    90→/// JS `Array.prototype.join` element coercion (null/undefined → "", nested

    92→    92→fn join_element(v: &Value) -> String {
    93→    93→    match v {
    94→    94→        Value::Null => String::new(),
    95→    95→        Value::String(s) => s.clone(),
    96→    96→        Value::Number(n) => n.to_string(),
    97→    97→        Value::Bool(b) => b.to_string(),
    98→    98→        Value::Array(items) => items
    99→    99→            .iter()
   100→   100→            .map(join_element)
   101→   101→            .collect::<Vec<_>>()
   102→   102→            .join(","),
   103→   103→        Value::Object(_) => "[object Object]".into(),
   104→   104→    }
   105→   105→}

   126→   126→/// Howard Hinnant civil_from_days (days since 1970-01-01 → y/m/d) — the
   127→   127→/// inverse of days_from_civil, valid across the whole i64 range.
   128→   128→fn civil_from_days(z: i64) -> (i64, u32, u32) {
   129→   129→    let z = z + 719_468;
   130→   130→    let era = if z >= 0 { z } else { z - 146_096 } / 146_097;
   131→   131→    let doe = z - era * 146_097; // [0, 146096]
   132→   132→    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365; // [0, 399]
   133→   133→    let y = yoe + era * 400;
   134→   134→    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100); // [0, 365]
   135→   135→    let mp = (5 * doy + 2) / 153; // [0, 11]
   136→   136→    let d = (doy - (153 * mp + 2) / 5 + 1) as u32; // [1, 31]
   137→   137→    let m = if mp < 10 { mp + 3 } else { mp - 9 } as u32; // [1, 12]
   138→   138→    (y + i64::from(m <= 2), m, d)
   139→   139→}

   141→   141→/// JS Date.toISOString() from an epoch-ms time value. `None` ⇔ the JS Date
   142→   142→/// would be invalid (|t| > 8.64e15 ms): toISOString throws "Invalid time
   143→   143→/// value" there and the original route surfaced that as a 500.
   144→   144→pub fn iso_from_ms(t: i64) -> Option<String> {
   145→   145→    if t.unsigned_abs() > 8_640_000_000_000_000 {
   146→   146→        return None;
   147→   147→    }
   148→   148→    let days = t.div_euclid(86_400_000);
   149→   149→    let sod = t.rem_euclid(86_400_000); // 0..86_399_999 (ECMA modulo: sign of divisor)
   150→   150→    let ms = (sod % 1000) as u32;
   151→   151→    let sec = sod / 1000;
   152→   152→    let (y, mo, d) = civil_from_days(days);
   153→   153→    let (hh, mm, ss) = (sec / 3600, (sec % 3600) / 60, sec % 60);
   154→   154→    let date = if (0..=9999).contains(&y) {
   155→   155→        format!("{y:04}-{mo:02}-{d:02}")
   156→   156→    } else {
   157→   157→        // JS pads the year to 6 digits with an explicit sign outside 0..9999.
   158→   158→        let sign = if y < 0 { '-' } else { '+' };
   159→   159→        format!("{sign}{:06}", y.abs())
   160→   160→    };
   161→   161→    Some(format!("{date}T{hh:02}:{mm:02}:{ss:02}.{ms:03}Z"))
   162→   162→}

   164→   164→/// Drizzle renders a REAL (fractional seconds) as `new Date(v * 1000)` and
   165→   165→/// the engine extracts components per ECMA: SecFromTime = ⌊t/1000⌋,
   166→   166→/// msFromTime = t mod 1000 (floored modulo) truncated to whole ms by the
   167→   167→/// engine's int cast. Rebuilding the time value from those parts matches
   168→   168→/// V8 for every integral-ms Date, including negatives.
   169→   169→fn real_to_ms(f: f64) -> i64 {
   170→   170→    let t = f * 1000.0;
   171→   171→    if !t.is_finite() {
   172→   172→        return i64::MAX; // Date(NaN) → invalid → throws, mirrored as 500
   173→   173→    }
   174→   174→    let secs = (t / 1000.0).floor();
   175→   175→    let ms = t - secs * 1000.0; // 0 ≤ ms < 1000
   176→   176→    (secs * 1000.0 + ms.trunc()) as i64
   177→   177→}

   179→   179→/// A timestamp column cell → CSV text. Err ⇔ the TS route would have thrown
   180→   180→/// (TEXT in a timestamp column: `value * 1000` → NaN → invalid Date).
   181→   181→fn iso_cell(v: ValueRef<'_>) -> Result<String, ()> {
   182→   182→    Ok(match v {
   183→   183→        ValueRef::Null => String::new(), // TS `r.startedAt ?? ""` family
   184→   184→        ValueRef::Integer(i) => i
   185→   185→            .checked_mul(1000)
   186→   186→            .and_then(iso_from_ms)
   187→   187→            .ok_or(())?,
   188→   188→        ValueRef::Real(f) => iso_from_ms(real_to_ms(f)).ok_or(())?,
   189→   189→        ValueRef::Text(_) => return Err(()),
   190→   190→        ValueRef::Blob(_) => return Err(()),
   191→   191→    })
   192→   192→}

   306→   306→/// JS Date semantics on an owned ranking cell: `None` marks a value whose
   307→   307→/// `Date.toISOString()` would throw ("Invalid time value") — deferred to
   308→   308→/// render time because the TS fold stores Date objects and only VISIBLE
   309→   309→/// keywords ever render their winning row's checkedAt.
   310→   310→enum Epoch {
   311→   311→    Iso(String),
   312→   312→    Invalid,

   315→   315→fn epoch_cell(v: ValueRef<'_>) -> Epoch {
   316→   316→    match v {
   317→   317→        ValueRef::Null => Epoch::Iso(String::new()), // notNull column; drizzle null → ""
   318→   318→        ValueRef::Integer(i) => i
   319→   319→            .checked_mul(1000)
   320→   320→            .and_then(iso_from_ms)
   321→   321→            .map(Epoch::Iso)
   322→   322→            .unwrap_or(Epoch::Invalid),
   323→   323→        ValueRef::Real(f) => iso_from_ms(real_to_ms(f)).map_or(Epoch::Invalid, Epoch::Iso),
   324→   324→        // TEXT in a timestamp column: `value * 1000` → NaN → Invalid Date →
   325→   325→        // toISOString() throws RangeError at render.
   326→   326→        ValueRef::Text(_) | ValueRef::Blob(_) => Epoch::Invalid,
   327→   327→    }
   328→   328→}

   330→   330→/// Current UTC civil date, `new Date().toISOString().split("T")[0]` —
   331→   331→/// reuses the 14.8a epoch→ISO helper and takes the date-only part.
   332→   332→fn utc_today() -> String {
   333→   333→    let ms = std::time::SystemTime::now()
   334→   334→        .duration_since(std::time::UNIX_EPOCH)
   335→   335→        .map(|d| d.as_millis() as i64)
   336→   336→        .unwrap_or(0);
   337→   337→    match iso_from_ms(ms) {
   338→   338→        Some(iso) => iso.split('T').next().unwrap_or("").to_string(),
   339→   339→        None => String::new(),
   340→   340→    }
   341→   341→}

   343→   343→/// Pragmatic mirror of JS `a.localeCompare(b)` (default locale) for the
   344→   344→/// backlinks disavow domain sort — validated pairwise against Node's ICU on
   345→   345→/// a 370-string [A-Za-z0-9.-] corpus (139k ordered pairs, 0 mismatches):
   346→   346→///   1. primary: case-FOLD (ICU compares letters case-insensitively —
   347→   347→///      "a".localeCompare("B") === -1);
   348→   348→///   2. tertiary, per char: LOWERCASE before UPPERCASE (ICU root collation
   349→   349→///      orders "b" < "B"; the corpus test proves it);
   350→   350→///   3. original bytes (stable-sort tiebreak; JS sort is stable since
   351→   351→///      ES2019, Rust sort_by is stable — insertion order kept on 0).
   352→   352→/// Documented deviation: full ICU collation for NON-ASCII (é/ß/İ….) and for
   353→   353→/// exotic punctuation (ICU ranks '_' before '-' before '.', unlike ASCII
   354→   354→/// byte order) is NOT mirrored — realistic hostnames are ASCII with only
   355→   355→/// letters/digits/dots/hyphens (IDN lives in URLs as punycode), where the
   356→   356→/// mirror is exact. No ICU dependency is acceptable here; recorded honestly.
   357→   357→fn locale_cmp(a: &str, b: &str) -> std::cmp::Ordering {
   358→   358→    let (fa, fb) = (a.to_lowercase(), b.to_lowercase());
   359→   359→    if fa != fb {
   360→   360→        return fa.cmp(&fb);
   361→   361→    }
   362→   362→    for (ca, cb) in a.chars().zip(b.chars()) {
   363→   363→        let (ra, rb) = (u8::from(ca.is_uppercase()), u8::from(cb.is_uppercase()));
   364→   364→        if ra != rb {
   365→   365→            return ra.cmp(&rb);
   366→   366→        }
   367→   367→    }
   368→   368→    a.cmp(&b)
   369→   369→}

   371→   371→/// JS Number(value) for the reports pinned cell: drizzle's boolean column
   372→   372→/// maps driver values with `Number(value) === 1`, then the route does
   373→   373→/// `r.pinned ? "true" : "false"`.
   374→   374→fn js_number(v: ValueRef<'_>) -> f64 {
   375→   375→    match v {
   376→   376→        ValueRef::Null => 0.0, // Number(null) === 0
   377→   377→        ValueRef::Integer(i) => i as f64,
   378→   378→        ValueRef::Real(f) => f,
   379→   379→        ValueRef::Text(t) => std::str::from_utf8(t)
   380→   380→            .ok()
   381→   381→            .and_then(|s| {
   382→   382→                let s = s.trim(); // Number() trims whitespace
   383→   383→                if s.is_empty() {
   384→   384→                    Some(0.0)
   385→   385→                } else {
   386→   386→                    s.parse::<f64>().ok()
   387→   387→                }
   388→   388→            })
   389→   389→            .unwrap_or(f64::NAN),
   390→   390→        ValueRef::Blob(_) => 0.0, // Number(Uint8Array) === 0
   391→   391→    }
   392→   392→}

   417→   417→    let mut conds: Vec<String> = Vec::new();
   418→   418→    let mut params: Vec<Sqv> = Vec::new();
   419→   419→    if let Some(id) = client_id {
   420→   420→        // TS: Number.isInteger(id) && id > 0 ? eq(column, id) : inArray(column, [-1])
   421→   421→        // The proxy only sends integers; a non-positive one (0 / negative /
   422→   422→        // its -1 sentinel for unparseable raw params) is the always-false arm.
   423→   423→        if id > 0 {
   424→   424→            conds.push(format!("{column} = ?"));
   425→   425→            params.push(Sqv::Integer(id));
   426→   426→        } else {
   427→   427→            conds.push(format!("{column} IN (-1)"));
   428→   428→        }
   429→   429→    }
   430→   430→    if let Some(ids) = visible_client_ids(db, user_id) {
   431→   431→        if ids.is_empty() {
   432→   432→            conds.push(format!("{column} IN (-1)"));
   433→   433→        } else {
   434→   434→            let list = ids.iter().map(|_| "?").collect::<Vec<_>>().join(", ");
   435→   435→            conds.push(format!("{column} IN ({list})"));
   436→   436→            params.extend(ids.into_iter().map(Sqv::Integer));
   437→   437→        }
   438→   438→    }
   439→   439→    (conds, params)
   440→   440→}
   441→   441→
   442→   442→fn build_scope(
   443→   443→    db: &crate::db::Db,
   444→   444→    column: &str,
   445→   445→    user_id: Option<i64>,
   446→   446→    client_id: Option<i64>,
   447→   447→) -> Scope {
   448→   448→    let (conds, params) = build_scope_conds(db, column, user_id, client_id);
   449→   449→    let where_sql = if conds.is_empty() {
   450→   450→        String::new()
   451→   451→    } else {
   452→   452→        format!(" WHERE {}", conds.join(" AND "))
   453→   453→    };
   454→   454→    Scope { where_sql, params }
   455→   455→}
   456→   456→
   457→   457→fn body_ids(body: &Value) -> (Option<i64>, Option<i64>) {
   458→   458→    (
   459→   459→        body.get("userId").and_then(Value::as_i64),
   460→   460→        body.get("clientId").and_then(Value::as_i64),
   461→   461→    )
   462→   462→}

   465→   465→    db_error_response(msg)

   468→   468→/// Run one export query on the shared connection, mapping each row to its
   469→   469→/// CSV cells through `map_row`. Errors collapse to strings for the 500
   470→   470→/// envelope (bare sqlite message, or "Invalid time value" where the TS
   471→   471→/// route's Date.toISOString() would have thrown).
   472→   472→fn run_export(
   473→   473→    db: &crate::db::Db,
   474→   474→    sql: &str,
   475→   475→    params: &[Sqv],
   476→   476→    map_row: impl Fn(&rusqlite::Row<'_>) -> Result<Vec<String>, String>,
   477→   477→) -> Result<Vec<Vec<String>>, String> {
   478→   478→    let inner = db.with_conn(|c| {
   479→   479→        Ok((|| -> Result<Vec<Vec<String>>, String> {
   480→   480→            let mut stmt = c.prepare(sql).map_err(|e| e.to_string())?;
   481→   481→            let mut q = stmt
   482→   482→                .query(rusqlite::params_from_iter(params.iter()))
   483→   483→                .map_err(|e| e.to_string())?;
   484→   484→            let mut out: Vec<Vec<String>> = Vec::new();
   485→   485→            while let Some(r) = q.next().map_err(|e| e.to_string())? {
   486→   486→                out.push(map_row(r)?);
   487→   487→            }
   488→   488→            Ok(out)
   489→   489→        })())
   490→   490→    });
   491→   491→    inner.map_err(|e| e.to_string()).flatten()
   492→   492→}
